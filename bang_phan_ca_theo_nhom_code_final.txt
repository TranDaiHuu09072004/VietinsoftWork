<script>
(function(){
  const groups=['A1','A2','B1','B2','C1','C2','D1','D2'];
  const summaryCols=['Tổng ca ngày','Tổng ca hành chính','Tổng ca giữa','Tổng ca đêm','Số ngày nghỉ(Ba...)','Ngày công ...'];
  const table=document.getElementById('groupShiftTable');
  if(!table) return;
  const head=table.querySelector('thead tr');
  const body=table.querySelector('tbody');

  for(let d=1; d<=31; d++) head.insertAdjacentHTML('beforeend', `<th class="col-day">${d}</th>`);
  summaryCols.forEach((col,index)=>head.insertAdjacentHTML('beforeend', `<th class="${index>=4?'col-total-wide':'col-total'}">${col}</th>`));

  groups.forEach((group,index)=>{
    let html=`<tr class="${index===0?'is-active':''}"><td class="sticky-col col-stt">${index+1}</td><td class="sticky-col col-group">${group}</td>`;
    for(let d=1; d<=31; d++){
      const type=d%7===2?'is-off':d%7===3?'is-work':'';
      const tip=type==='is-off'?'Ngày nghỉ':type==='is-work'?'Ngày đi ca':'';
      html+=`<td class="col-day ${type}" ${tip?`data-tip="${tip}"`:''}></td>`;
    }
    summaryCols.forEach(()=>html+='<td></td>');
    body.insertAdjacentHTML('beforeend', html+'</tr>');
  });
})();
</script>
