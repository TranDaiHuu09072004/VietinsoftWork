(function(){
  const employees=[
    ['003','LÊ MINH TRÍ','Dev & implement'],['004','NGUYỄN MINH TRÍ','Dev & implement'],['005','PHẠM VĂN PHÁT','Dev & implement'],['006','HÀ PHI HÙNG','Dev & implement'],['007','PHÙNG TRUNG TÍN','Dev & implement'],['008','BÙI TRẦN THẮNG DUY','Dev & implement'],['011','NGUYỄN THANH LONG','R&D'],['012','NGUYỄN TRƯỜNG LONG','Dev & implement'],['013','NGUYỄN TẤN CHIẾN','Kinh doanh'],['014','VŨ ĐÌNH CƯỜNG','Nhân sự'],['029','TRẦN TRUNG HIẾU','Dev & implement'],['034','TRẦN KHÁNH LY','Kinh doanh'],['038','PHẠM THỊ THUÝ HƯỜNG','Marketing'],['040','NGUYỄN QUỐC LƯƠNG','Dev & implement'],['041','HỨA VĨNH ĐỨC','Dev & implement'],['045','NGUYỄN THẮNG','Dev & implement'],['046','ĐẶNG MINH TÀI','R&D'],['047','NGUYỄN VĂN THỐNG','Dev & implement'],['052','LÊ TRẦN HOÀNG NAM','Kinh doanh'],['054','NGUYỄN THỊ THANH TÂM','Kinh doanh'],['060','LÊ VĂN TỨ','R&D'],['061','THẠCH HOÀNG THÀNH','R&D'],['062','PHẠM MINH TRÍ','Dev & implement'],['063','NGUYỄN TỬ ÁNH TUYẾT','Kinh doanh'],['065','NGUYỄN NGỌC MINH','Thực tập'],['066','HUỲNH NGỌC PHÚC','Thực tập']
  ];
  const table=document.getElementById('shiftMonitorTable');
  if(!table) return;
  const head=table.querySelector('thead tr');
  const body=table.querySelector('tbody');
  for(let d=1; d<=30; d++) head.insertAdjacentHTML('beforeend', `<th class="col-day">${d}</th>`);
  employees.forEach((emp,rowIndex)=>{
    let html=`<tr class="${rowIndex===0?'is-active':''}"><td class="sticky-col col-code">${emp[0]}</td><td class="sticky-col col-name">${emp[1]}</td><td class="col-dept">${emp[2]}</td>`;
    for(let d=1; d<=30; d++){
      const type=d%7===2?'is-off':d%7===3?'is-work':'';
      const tip=type==='is-off'?'Ngày nghỉ':type==='is-work'?'Ngày đi ca':'';
      html+=`<td class="col-day ${type}" ${tip?`data-tip="${tip}"`:''}></td>`;
    }
    body.insertAdjacentHTML('beforeend', html+'</tr>');
  });
})();
</script>
